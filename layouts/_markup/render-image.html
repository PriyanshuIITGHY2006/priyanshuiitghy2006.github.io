{{- /* Images in posts. A page resource or a file under assets/ (static media is mounted there) gets resized WebP copies; anything else is linked as written, made site-relative so it works from any page. */ -}} {{- $u := urls.Parse .Destination -}} {{- $res := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "/" (strings.TrimPrefix "./" $u.Path) -}} {{- $res = or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- end -}} {{- if and $res (in (slice "jpeg" "png" "webp" "gif" "bmp" "tiff") $res.MediaType.SubType) -}} {{- partial "responsive-img.html" (dict "img" $res "alt" .Text "sizes" "(min-width: 768px) 720px, 100vw") -}} {{- else -}} {{- $src := $u.String -}} {{- if not $u.IsAbs }}{{ $src = relURL (strings.TrimPrefix "/" $src) }}{{ end -}} {{- end -}} {{- /**/ -}}
